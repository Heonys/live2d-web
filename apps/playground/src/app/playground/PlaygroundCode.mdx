```tsx title="Avatar.tsx"
'use client'

import type { Live2DModelController } from 'live2d-web/react'
import { Live2DCanvas, Live2DModel } from 'live2d-web/react'
import { useState } from 'react'

export function Avatar() {
  const [controller, setController]
    = useState<Live2DModelController | null>(null)

  return (
    <Live2DCanvas
      coreUrl="/live2dcubismcore.min.js"
      quality="auto"
    >
      <Live2DModel
        src="/models/avatar.model3.json"
        fit="upper-body"
        followPointer
        onLoad={setController}
        onTap={(hitAreas) => {
          if (hitAreas.includes('Body'))
            void controller?.motion('Tap')
        }}
      />
    </Live2DCanvas>
  )
}
```
